
package com.example.altyazicevir.merger

import android.os.SystemClock
import java.text.Normalizer
import java.util.LinkedHashMap
import java.util.Locale
import kotlin.math.abs
import kotlin.math.max
import kotlin.math.min

// ==========================================
// 1. TEXT NORMALIZER
// ==========================================
object TextNormalizer {

    private val punctuationRegex = Regex("[^\\p{L}\\p{N}\\s]")
    private val multiSpaceRegex = Regex("\\s+")

    fun normalize(text: String): String {
        return Normalizer.normalize(text, Normalizer.Form.NFC)
            .lowercase(Locale.forLanguageTag("tr"))
            .replace(punctuationRegex, " ")
            .replace(multiSpaceRegex, " ")
            .trim()
    }
}

// ==========================================
// 2. SIMILARITY ENGINE
// Gerçek Banded Levenshtein
// ==========================================
class SimilarityEngine(
    private val similarityThreshold: Double
) {

    init {
        require(similarityThreshold in 0.0..1.0) {
            "similarityThreshold 0.0 ile 1.0 arasında olmalıdır"
        }
    }

    fun calculateSimilarity(s1: String, s2: String): Double {
        if (s1 == s2) return 1.0

        val maxLen = max(s1.length, s2.length)
        if (maxLen == 0) return 1.0

        val maxAllowedDistance =
            (maxLen * (1.0 - similarityThreshold)).toInt()

        if (abs(s1.length - s2.length) > maxAllowedDistance) {
            return 0.0
        }

        val distance = bandedLevenshteinDistance(
            lhs = s1,
            rhs = s2,
            maxDistance = maxAllowedDistance
        )

        if (distance > maxAllowedDistance) return 0.0

        return 1.0 - distance.toDouble() / maxLen
    }

    private fun bandedLevenshteinDistance(
        lhs: CharSequence,
        rhs: CharSequence,
        maxDistance: Int
    ): Int {

        val lhsLength = lhs.length
        val rhsLength = rhs.length

        if (abs(lhsLength - rhsLength) > maxDistance) {
            return Int.MAX_VALUE
        }

        val inf = Int.MAX_VALUE / 4

        var previous = IntArray(lhsLength + 1) { inf }
        var current = IntArray(lhsLength + 1) { inf }

        for (j in 0..min(lhsLength, maxDistance)) {
            previous[j] = j
        }

        for (i in 1..rhsLength) {

            current.fill(inf)

            if (i <= maxDistance) {
                current[0] = i
            }

            val start = max(1, i - maxDistance)
            val end = min(lhsLength, i + maxDistance)

            for (j in start..end) {

                val substitutionCost =
                    if (lhs[j - 1] == rhs[i - 1]) 0 else 1

                current[j] = minOf(
                    current[j - 1] + 1,
                    previous[j] + 1,
                    previous[j - 1] + substitutionCost
                )
            }

            val temp = previous
            previous = current
            current = temp
        }

        return previous[lhsLength]
    }
}

// ==========================================
// 3. SUBTITLE CANDIDATE
// ==========================================
private data class VariantRecord(
    var rawText: String,
    var frequency: Int,
    var lastSeenTime: Long
)

data class SubtitleCandidate(
    var rawText: String,
    var normalizedText: String,
    val firstSeenTime: Long,
    var lastSeenTime: Long,
    var hitCount: Int,
    val variantFrequencies: MutableMap<String, VariantRecord> =
        mutableMapOf(),
    var hasEmitted: Boolean = false,
    var lastEmittedNormalized: String? = null
) {

    fun addVariant(raw: String, now: Long) {

        val normalized = TextNormalizer.normalize(raw)
        if (normalized.isBlank()) return

        val existing = variantFrequencies[normalized]

        if (existing == null) {
            variantFrequencies[normalized] = VariantRecord(
                rawText = raw,
                frequency = 1,
                lastSeenTime = now
            )
        } else {
            existing.frequency++
            existing.lastSeenTime = now

            if (raw.length > existing.rawText.length) {
                existing.rawText = raw
            }
        }
    }

    fun observe(raw: String, now: Long) {

        hitCount++
        lastSeenTime = now

        addVariant(raw, now)

        val normalized = TextNormalizer.normalize(raw)

        // Aday metnini yalnızca daha uzun okuma geldiğinde büyüt.
        if (normalized.length > normalizedText.length) {
            rawText = raw
            normalizedText = normalized
        }
    }

    fun getBestVariantText(): String {

        return variantFrequencies.entries
            .maxWithOrNull(
                compareBy<Map.Entry<String, VariantRecord>> {
                    it.key.length
                }.thenBy {
                    it.value.frequency
                }.thenBy {
                    it.value.lastSeenTime
                }
            )
            ?.value
            ?.rawText
            ?: rawText
    }
}

// ==========================================
// 4. EMISSION CACHE
// TTL + LRU
// ==========================================
class EmissionCache(
    private val ttlMillis: Long,
    private val maxSize: Int
) {

    init {
        require(ttlMillis > 0) {
            "emissionCacheTtlMillis 0'dan büyük olmalıdır"
        }

        require(maxSize > 0) {
            "maxCacheSize 0'dan büyük olmalıdır"
        }
    }

    private val cache =
        LinkedHashMap<String, Long>(maxSize, 0.75f, true)

    fun contains(normalizedText: String, now: Long): Boolean {
        prune(now)

        // get() erişim sırasını günceller ve LRU davranışını korur.
        return cache[normalizedText] != null
    }

    fun put(normalizedText: String, now: Long) {

        cache[normalizedText] = now

        while (cache.size > maxSize) {
            val oldestKey = cache.keys.iterator().next()
            cache.remove(oldestKey)
        }
    }

    private fun prune(now: Long) {

        val iterator = cache.entries.iterator()

        while (iterator.hasNext()) {
            val entry = iterator.next()

            if (now - entry.value >= ttlMillis) {
                iterator.remove()
            }
        }
    }
}

// ==========================================
// 5. FRAGMENT MERGER V8
// ==========================================
class FragmentMerger(
    private val minConsecutiveMatches: Int = 3,
    private val minStabilityMillis: Long = 400L,
    private val maxWaitMillis: Long = 1500L,
    private val maxInactivityMillis: Long = 600L,
    private val similarityThreshold: Double = 0.85,
    emissionCacheTtlMillis: Long = 10000L,
    maxCacheSize: Int = 100,
    private val emptyGracePeriodMillis: Long = 500L,
    private val clock: () -> Long = {
        SystemClock.elapsedRealtime()
    }
) {

    init {
        require(minConsecutiveMatches > 0)
        require(minStabilityMillis >= 0)
        require(maxWaitMillis > 0)
        require(maxInactivityMillis > 0)
        require(similarityThreshold in 0.0..1.0)
        require(emptyGracePeriodMillis >= 0)
    }

    private var currentCandidate: SubtitleCandidate? = null
    private var pendingAlternativeCandidate: SubtitleCandidate? = null

    private var emptySinceTimestamp: Long = -1L

    private val similarityEngine =
        SimilarityEngine(similarityThreshold)

    private val emissionCache =
        EmissionCache(emissionCacheTtlMillis, maxCacheSize)

    private val lock = Any()

    sealed class MergerResult {

        object Pending : MergerResult()

        data class Stable(
            val text: String,
            val isUpdate: Boolean = false
        ) : MergerResult()
    }

    // ==========================================
    // MAIN OCR INPUT
    // ==========================================
    fun processRawOcr(rawText: String): MergerResult =
        synchronized(lock) {

            val now = clock()
            val trimmedRaw = rawText.trim()
            val cleaned = TextNormalizer.normalize(trimmedRaw)

            // 1. Boş OCR kontrolü
            if (cleaned.isBlank()) {

                val current = currentCandidate

                if (current != null) {

                    if (emptySinceTimestamp == -1L) {
                        emptySinceTimestamp = now
                    }

                    if (
                        now - emptySinceTimestamp >=
                        emptyGracePeriodMillis
                    ) {
                        val result = finalizeCandidate(current, now)
                        resetStateInternal()
                        return@synchronized result
                    }
                }

                return@synchronized checkTimeoutInternal(now)
            }

            emptySinceTimestamp = -1L

            // 2. Önceki adayın zaman aşımını kontrol et.
            val timeoutResult = checkTimeoutInternal(now)

            val current = currentCandidate

            // 3. Aktif aday yoksa yeni aday oluştur.
            if (current == null) {
                initCandidate(trimmedRaw, cleaned, now)
                return@synchronized timeoutResult
            }

            val similarity = similarityEngine.calculateSimilarity(
                current.normalizedText,
                cleaned
            )

            // 4. Mevcut altyazının devamı mı?
            if (
                isExtensionOrSimilar(
                    current.normalizedText,
                    cleaned,
                    similarity
                )
            ) {

                current.observe(trimmedRaw, now)
                pendingAlternativeCandidate = null

                if (isValidForEmission(current)) {
                    return@synchronized emitCandidate(current, now)
                }

                return@synchronized timeoutResult
            }

            // 5. Farklı metin: alternatif aday yönetimi.
            val pendingAlt = pendingAlternativeCandidate

            val altSimilarity = if (pendingAlt != null) {
                similarityEngine.calculateSimilarity(
                    pendingAlt.normalizedText,
                    cleaned
                )
            } else {
                0.0
            }

            if (
                pendingAlt != null &&
                isExtensionOrSimilar(
                    pendingAlt.normalizedText,
                    cleaned,
                    altSimilarity
                )
            ) {
                pendingAlt.observe(trimmedRaw, now)
            } else {

                val newAlt = SubtitleCandidate(
                    rawText = trimmedRaw,
                    normalizedText = cleaned,
                    firstSeenTime = now,
                    lastSeenTime = now,
                    hitCount = 1
                )

                newAlt.addVariant(trimmedRaw, now)
                pendingAlternativeCandidate = newAlt
            }

            val alternative = pendingAlternativeCandidate!!

            // 6. Alternatif aday gerçekten kararlı mı?
            if (isValidForEmission(alternative)) {

                val result = finalizeCandidate(current, now)

                // Alternatif adayı ana aday hâline getir.
                currentCandidate = alternative
                pendingAlternativeCandidate = null
                emptySinceTimestamp = -1L

                return@synchronized result
            }

            return@synchronized timeoutResult
        }

    // ==========================================
    // FLUSH
    // ==========================================
    fun flush(): MergerResult = synchronized(lock) {

        val now = clock()
        val current = currentCandidate

        if (
            current != null &&
            emptySinceTimestamp != -1L &&
            now - emptySinceTimestamp >= emptyGracePeriodMillis
        ) {

            val result = finalizeCandidate(current, now)
            resetStateInternal()

            return@synchronized result
        }

        checkTimeoutInternal(now)
    }

    // ==========================================
    // TIMEOUT MANAGEMENT
    // ==========================================
    private fun checkTimeoutInternal(
        now: Long
    ): MergerResult {

        val current = currentCandidate
            ?: return MergerResult.Pending

        // İlk yayın öncesinde toplam bekleme süresi uygulanır.
        val totalLifeExceeded =
            !current.hasEmitted &&
            now - current.firstSeenTime >= maxWaitMillis

        // Yayınlanmış altyazılar sessizlikte tamamlanır.
        val inactivityExceeded =
            now - current.lastSeenTime >= maxInactivityMillis

        if (totalLifeExceeded || inactivityExceeded) {

            val result = finalizeCandidate(current, now)
            resetStateInternal()

            return result
        }

        return MergerResult.Pending
    }

    // ==========================================
    // CANDIDATE VALIDATION
    // ==========================================
    private fun isValidForEmission(
        candidate: SubtitleCandidate
    ): Boolean {

        val stableEnough =
            candidate.hitCount >= minConsecutiveMatches

        val timeEnough =
            candidate.lastSeenTime -
                candidate.firstSeenTime >= minStabilityMillis

        return stableEnough && timeEnough
    }

    // ==========================================
    // EMISSION
    // ==========================================
    private fun emitCandidate(
        candidate: SubtitleCandidate,
        now: Long
    ): MergerResult {

        val text = candidate.getBestVariantText()
        val normalized = TextNormalizer.normalize(text)

        if (normalized.isBlank()) {
            return MergerResult.Pending
        }

        // Aynı sürümü tekrar gönderme.
        if (normalized == candidate.lastEmittedNormalized) {
            return MergerResult.Pending
        }

        val isUpdate = candidate.hasEmitted

        // Yeni altyazılarda cache kontrolü uygula.
        // Aynı altyazının güncellemeleri cache tarafından engellenmez.
        if (!isUpdate && emissionCache.contains(normalized, now)) {
            candidate.hasEmitted = true
            candidate.lastEmittedNormalized = normalized
            return MergerResult.Pending
        }

        if (!isUpdate) {
            emissionCache.put(normalized, now)
        }

        candidate.hasEmitted = true
        candidate.lastEmittedNormalized = normalized

        return MergerResult.Stable(
            text = text,
            isUpdate = isUpdate
        )
    }

    // ==========================================
    // FINALIZE
    // ==========================================
    private fun finalizeCandidate(
        candidate: SubtitleCandidate,
        now: Long
    ): MergerResult {

        if (!isValidForEmission(candidate)) {
            return MergerResult.Pending
        }

        return emitCandidate(candidate, now)
    }

    // ==========================================
    // INIT / RESET
    // ==========================================
    private fun initCandidate(
        raw: String,
        normalized: String,
        now: Long
    ) {

        val candidate = SubtitleCandidate(
            rawText = raw,
            normalizedText = normalized,
            firstSeenTime = now,
            lastSeenTime = now,
            hitCount = 1
        )

        candidate.addVariant(raw, now)

        currentCandidate = candidate
        pendingAlternativeCandidate = null
        emptySinceTimestamp = -1L
    }

    private fun resetStateInternal() {

        currentCandidate = null
        pendingAlternativeCandidate = null
        emptySinceTimestamp = -1L
    }

    // ==========================================
    // SIMILARITY / EXTENSION
    // ==========================================
    private fun isExtensionOrSimilar(
        oldText: String,
        newText: String,
        similarity: Double
    ): Boolean {

        if (oldText == newText) return true

        // Gerçek metin uzatması.
        if (newText.startsWith(oldText)) return true

        // OCR'nin geçici olarak daha kısa okuması.
        if (oldText.startsWith(newText)) return true

        // Benzerlik eşiği.
        return similarity >= similarityThreshold
    }
}
